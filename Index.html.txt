<script>
let teams = Array.from({length:16}, (_,i)=>({
  name: "Team " + (i+1),
  img: "team" + (i+1) + ".png"
}));

function changeTeam(element) {
  event.stopPropagation(); // Prevent advancing when clicking
  const newImg = prompt("Enter URL of new team image:");
  const newName = prompt("Enter new team name:");
  if (newImg) {
    element.querySelector("img").src = newImg;
    teams[element.parentElement.dataset.index].img = newImg;
  }
  if (newName) {
    element.querySelector("p").textContent = newName;
    teams[element.parentElement.dataset.index].name = newName;
  }
}

// Advance winner from a match to next round
function advanceWinner(index1,index2,nextRoundLength) {
  const team1 = teams[index1];
  const team2 = teams[index2];
  const winnerName = prompt(`Who wins?\n1: ${team1.name}\n2: ${team2.name}`);
  let winner = (winnerName == "1") ? team1 : team2;
  
  const nextRound = Math.floor(index1/2);
  teams[nextRoundLength + nextRound] = winner; // Add to next round array
  
  renderBracket();
}

// Render the bracket visually
function renderBracket() {
  function createRoundHTML(start, length, id) {
    const roundDiv = document.getElementById(id);
    roundDiv.innerHTML = "";
    for(let i=start;i<start+length;i+=2){
      if(!teams[i] || !teams[i+1]) continue;
      const matchDiv = document.createElement("div");
      matchDiv.className = "match";
      matchDiv.dataset.index = i;

      [i,i+1].forEach(idx=>{
        const teamDiv = document.createElement("div");
        teamDiv.className = "team";
        teamDiv.onclick = ()=>changeTeam(teamDiv);
        const img = document.createElement("img");
        img.src = teams[idx].img;
        const p = document.createElement("p");
        p.textContent = teams[idx].name;
        teamDiv.appendChild(img);
        teamDiv.appendChild(p);
        matchDiv.appendChild(teamDiv);
      });
      roundDiv.appendChild(matchDiv);
    }
  }
  createRoundHTML(0,16,"round1");
  createRoundHTML(16,8,"round2");
  createRoundHTML(24,4,"semifinals");
  createRoundHTML(28,2,"finals");
  if(teams[30]){
    const champDiv = document.getElementById("champion");
    champDiv.innerHTML = `<div class="team winner"><img src="${teams[30].img}"><p>${teams[30].name}</p></div>`;
  }
}

renderBracket();
</script>
